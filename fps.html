const {WebSocketServer}=require("ws");const wss=new WebSocketServer({port:8080});
wss.on("connection",ws=>{ws.on("message",d=>{let m;try{m=JSON.parse(d)}catch{return}
if(m.t==="join"){ws.room=m.room;ws.pid=m.id}
for(const c of wss.clients)if(c!==ws&&c.readyState===1&&c.room===ws.room)c.send(String(d))});
ws.on("close",()=>{for(const c of wss.clients)if(c.readyState===1&&c.room===ws.room)c.send(JSON.stringify({t:"leave",id:ws.pid}))})});
